cmake_minimum_required(VERSION 3.21)
project(ShutterDB VERSION 0.2.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
option(SHUTTER_BUILD_TESTS "Build tests" ${PROJECT_IS_TOP_LEVEL})
option(SHUTTER_BUILD_CLI "Build shutter CLI" ${PROJECT_IS_TOP_LEVEL})
option(SHUTTER_BUILD_BENCHMARKS "Build benchmarks" OFF)
option(SHUTTER_BUILD_COMPARISON "Build the asset-cache comparison" OFF)
option(SHUTTER_COMPARE_ROCKSDB "Include RocksDB in the asset-cache comparison" OFF)
option(SHUTTER_BUILD_ASSET_CACHE "Build the preview cache (requires OpenSSL, PNG and JPEG)" OFF)
option(SHUTTER_BUILD_FUZZERS "Build Clang libFuzzer targets" OFF)
option(SHUTTER_BUILD_STRESS "Build stress test" OFF)
option(SHUTTER_ASAN "Enable AddressSanitizer" OFF)
option(SHUTTER_UBSAN "Enable UndefinedBehaviorSanitizer" OFF)
option(SHUTTER_CLANG_TIDY "Run clang-tidy while building" OFF)
if(SHUTTER_CLANG_TIDY)
  find_program(SHUTTER_TIDY NAMES clang-tidy REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${SHUTTER_TIDY};--warnings-as-errors=*")
  set(CMAKE_C_CLANG_TIDY "${SHUTTER_TIDY};--warnings-as-errors=*")
endif()
if(SHUTTER_ASAN)
  if(MSVC)
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO)
  else()
    add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address)
  endif()
endif()
if(SHUTTER_UBSAN)
  if(MSVC)
    message(FATAL_ERROR "Use GCC or Clang on Linux for UBSan")
  endif()
  add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=undefined)
endif()
set(SHUTTER_SOURCES src/db.cpp src/cache.cpp src/storage.cpp src/record.cpp src/checksum.cpp src/compaction.cpp src/verify.cpp)
function(shutter_configure target)
  target_compile_features(${target} PUBLIC cxx_std_20)
  target_link_libraries(${target} PUBLIC Threads::Threads)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endfunction()
add_library(ShutterDB STATIC ${SHUTTER_SOURCES})
add_library(ShutterDB::ShutterDB ALIAS ShutterDB)
shutter_configure(ShutterDB)
set_target_properties(ShutterDB PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(SHUTTER_BUILD_CLI)
  add_executable(shutter apps/shutter.cpp)
  target_link_libraries(shutter PRIVATE ShutterDB::ShutterDB)
  shutter_configure(shutter)
  install(TARGETS shutter RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(SHUTTER_BUILD_TESTS)
  enable_testing()
  add_library(shutter_test_core STATIC ${SHUTTER_SOURCES})
  shutter_configure(shutter_test_core)
  target_compile_definitions(shutter_test_core PUBLIC SHUTTER_TESTING)
  add_executable(shutter_tests tests/test_db.cpp tests/test_cache.cpp)
  target_include_directories(shutter_tests PRIVATE src)
  target_include_directories(shutter_tests SYSTEM PRIVATE third_party)
  target_link_libraries(shutter_tests PRIVATE shutter_test_core)
  shutter_configure(shutter_tests)
  add_test(NAME engine COMMAND shutter_tests)
  set_tests_properties(engine PROPERTIES TIMEOUT 180)
  add_executable(shutter_fuzz_smoke fuzz/smoke.cpp)
  target_include_directories(shutter_fuzz_smoke PRIVATE src fuzz)
  target_link_libraries(shutter_fuzz_smoke PRIVATE ShutterDB)
  shutter_configure(shutter_fuzz_smoke)
  add_test(NAME parser_smoke COMMAND shutter_fuzz_smoke)
  add_executable(shutter_example examples/basic.cpp)
  target_link_libraries(shutter_example PRIVATE ShutterDB)
  add_test(NAME public_api_example COMMAND shutter_example)
  if(SHUTTER_BUILD_CLI)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
      add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:shutter>)
      add_test(NAME corruption COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_corruption.py $<TARGET_FILE:shutter>)
      if(SHUTTER_BUILD_ASSET_CACHE)
        add_test(NAME asset_cache COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_asset_cache.py $<TARGET_FILE:shutter_asset_cache> $<TARGET_FILE:shutter>)
      endif()
      set_tests_properties(corruption PROPERTIES TIMEOUT 120)
      if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT SHUTTER_ASAN)
        add_library(shutter_fault_io SHARED tests/fault_io.cpp)
        target_compile_features(shutter_fault_io PRIVATE cxx_std_20)
        target_link_libraries(shutter_fault_io PRIVATE ${CMAKE_DL_LIBS})
        add_test(NAME syscall_faults COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_io_faults.py $<TARGET_FILE:shutter> $<TARGET_FILE:shutter_fault_io>)
        set_tests_properties(syscall_faults PROPERTIES TIMEOUT 120)
      endif()
    endif()
  endif()
endif()
if(SHUTTER_BUILD_STRESS)
  add_executable(shutter_stress tests/stress.cpp)
  target_link_libraries(shutter_stress PRIVATE ShutterDB)
  shutter_configure(shutter_stress)
endif()
if(SHUTTER_BUILD_BENCHMARKS)
  add_executable(shutter_bench benchmarks/bench.cpp)
  target_link_libraries(shutter_bench PRIVATE ShutterDB)
  shutter_configure(shutter_bench)
endif()
if(SHUTTER_BUILD_COMPARISON)
  find_package(SQLite3 REQUIRED)
  add_executable(shutter_compare benchmarks/asset_cache.cpp)
  target_link_libraries(shutter_compare PRIVATE ShutterDB::ShutterDB SQLite::SQLite3)
  if(SHUTTER_COMPARE_ROCKSDB)
    find_package(RocksDB CONFIG REQUIRED)
    target_link_libraries(shutter_compare PRIVATE RocksDB::rocksdb)
    target_compile_definitions(shutter_compare PRIVATE SHUTTER_HAVE_ROCKSDB)
  endif()
  shutter_configure(shutter_compare)
endif()
if(SHUTTER_BUILD_ASSET_CACHE)
  enable_language(C)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  find_package(PNG 1.6 REQUIRED)
  find_package(JPEG REQUIRED)
  add_executable(shutter_asset_cache examples/asset_cache.cpp examples/jpeg_decode.c)
  target_compile_features(shutter_asset_cache PRIVATE c_std_11)
  target_link_libraries(shutter_asset_cache PRIVATE ShutterDB::ShutterDB OpenSSL::Crypto PNG::PNG JPEG::JPEG)
  shutter_configure(shutter_asset_cache)
endif()
if(SHUTTER_BUILD_FUZZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR MSVC)
    message(FATAL_ERROR "libFuzzer targets require Clang on a POSIX platform")
  endif()
  add_library(shutter_fuzz_core STATIC ${SHUTTER_SOURCES})
  shutter_configure(shutter_fuzz_core)
  target_compile_options(shutter_fuzz_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  foreach(mode IN ITEMS record scanner recovery verify)
    add_executable(fuzz_${mode} fuzz/target.cpp)
    target_compile_definitions(fuzz_${mode} PRIVATE SHUTTER_FUZZ_${mode})
    target_compile_options(fuzz_${mode} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(fuzz_${mode} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_include_directories(fuzz_${mode} PRIVATE src fuzz)
    target_link_libraries(fuzz_${mode} PRIVATE shutter_fuzz_core)
  endforeach()
endif()

write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ShutterDBConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ShutterDBConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/ShutterDBConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ShutterDB)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/ShutterDB)
